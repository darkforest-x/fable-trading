//@version=5
indicator("Range Fanshen V2.1 1H Vertical Rays", overlay=true, max_lines_count=500)

// 1. Inputs Matching Settings Specimen
show_rays = input.bool(true, "显示 1H 勾叉竖射线", group="Ray Settings")
ray_color = input.color(#F23645, "射线颜色", group="Ray Settings")
ray_width = input.int(1, "线宽 (px)", minval=1, maxval=4, group="Ray Settings")
max_rays  = input.int(100, "保留最近信号数", minval=10, maxval=200, group="Ray Settings")

// 2. Anchor to Fixed 1-Hour Timeframe
is_1h_bar = timeframe.isintraday and (timeframe.multiplier == 60)
[h1_time, h1_low, h1_high, bull_sig, bear_sig] = request.security(
    syminfo.tickerid, "60", 
    [time, low, high, bull_cond, bear_cond], 
    lookahead=barmerge.lookahead_off
)

// 3. Unidirectional Rays: Low -> Upwards (Bull), High -> Downwards (Bear)
if show_rays and bull_sig
    // Anchored strictly at candle open timestamp, extends upwards
    line.new(x1=h1_time, y1=h1_low, x2=h1_time, y2=h1_low + 1000000, xloc=xloc.bar_time, extend=extend.none, color=ray_color, width=ray_width)

if show_rays and bear_sig
    // Anchored strictly at candle open timestamp, extends downwards
    line.new(x1=h1_time, y1=h1_high, x2=h1_time, y2=0, xloc=xloc.bar_time, extend=extend.none, color=ray_color, width=ray_width)
✓ 零水平位移: 以 `h1_time` 时间戳锚定，换周期不漂移。
✓ 单向延伸: 勾从 Low 向上贯穿；叉从 High 向下打底。
✓ 附带参数: 默认开启，支持颜色、线宽与 100 条环形缓冲。
